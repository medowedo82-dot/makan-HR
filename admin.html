async function submitNewEmployee() { const name = document.getElementById('fullName').value.trim(); const nId = document.getElementById('nationalId').value.trim(); const dept = document.getElementById('department').value.trim(); const shift = document.getElementById('shiftType').value; const msg = document.getElementById('adminMsg'); if (!name || !nId || !dept) { msg.textContent = 'يرجى ملء جميع الحقول المطلوبة.'; msg.className = 'status-box error'; msg.style.display = 'block'; return; } msg.textContent = 'جاري تسجيل الموظف وتوليد البيانات...'; msg.className = 'status-box'; msg.style.display = 'block'; try { const res = await fetch(SCRIPT_URL, { method: 'POST', redirect: 'follow', // ضروري لتمكين المتصفح من متابعة تحويل Google واستلام البيانات headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: JSON.stringify({ action: 'createEmployee', fullName: name, nationalId: nId, department: dept, shiftType: shift }) }); const data = await res.json(); if (data && data.success && data.result) { const emp = data.result; msg.className = 'status-box success'; msg.innerHTML = `
✅ تم إنشاء الحساب بنجاح!
اسم الموظف:${emp.name}
كود الدخول:${emp.code}
كلمة المرور:${emp.password}
`; document.getElementById('fullName').value = ''; document.getElementById('nationalId').value = ''; document.getElementById('department').value = ''; } else { msg.textContent = (data && data.error) ? data.error : 'تعذر إتمام العملية.'; msg.className = 'status-box error'; } } catch (e) { msg.textContent = 'حدث خطأ في قراءة الرد. يرجى مراجعة شيت Employees للتأكد من تسجيل الحساب.'; msg.className = 'status-box error'; } }
